const tg = window.Telegram?.WebApp; if (tg) { tg.ready(); tg.expand(); } const $ = (id) => document.getElementById(id); let projects = []; let selectedId = null; let busy = false; function setMessage(message, type = "") { $("message").textContent = message; $("message").className = "notice " + type; } function setBusy(value) { busy = value; for (const id of [ "create-btn", "save-btn", "reload-btn", "delete-btn" ]) { $(id).disabled = value; } } async function api(path, options = {}) { const initData = tg?.initData || ""; if (!initData) { throw new Error( "Открой DevPocket через кнопку бота в Telegram, " + "чтобы подтвердить аккаунт." ); } const headers = { "X-Telegram-Init-Data": initData, ...(options.body ? {"Content-Type": "application/json"} : {}), ...(options.headers || {}) }; const response = await fetch(path, { ...options, headers }); let data; try { data = await response.json(); } catch { throw new Error("Сервер вернул некорректный ответ."); } if (!response.ok) { const detail = typeof data.detail === "string" ? data.detail : "Ошибка запроса (" + response.status + ")"; throw new Error(detail); } return data; } function renderProjects() { const list = $("project-list"); list.replaceChildren(); for (const project of projects) { const button = document.createElement("button"); button.className = "project-item" + (project.id === selectedId ? " active" : ""); button.type = "button"; const label = document.createElement("span"); label.className = "project-label"; label.textContent = project.name; const details = document.createElement("small"); details.textContent = project.language + " · " + new Date(project.updated_at).toLocaleString(); label.appendChild(details); const symbol = document.createElement("span"); symbol.textContent = "📄"; button.append(symbol, label); button.addEventListener("click", () => openProject(project.id)); list.appendChild(button); } $("empty").textContent = projects.length ? "Выбери проект, чтобы открыть его." : "Проектов пока нет. Создай первый выше."; $("project-id").textContent = selectedId === null ? "Не выбран" : "ID " + selectedId; } function clearEditor() { selectedId = null; $("project-name").value = ""; $("language").value = "python"; $("code").value = ""; renderProjects(); } async function loadProjects() { const data = await api("/api/projects"); projects = data.projects; if (selectedId !== null && !projects.some(p => p.id === selectedId)) { clearEditor(); } else { renderProjects(); } $("connection").textContent = "● API подключён"; } async function openProject(id) { if (busy) return; setBusy(true); try { const data = await api("/api/projects/" + id); const project = data.project; selectedId = project.id; $("project-name").value = project.name; $("language").value = project.language; $("code").value = project.content; renderProjects(); setMessage("Проект открыт.", "success"); } catch (error) { setMessage(error.message, "error"); } finally { setBusy(false); } } async function createNewProject() { if (busy) return; const name = $("new-name").value.trim(); if (!name) { setMessage("Сначала введи название проекта.", "error"); return; } setBusy(true); try { const data = await api("/api/projects", { method: "POST", body: JSON.stringify({ name, language: "python", content: "" }) }); $("new-name").value = ""; await loadProjects(); selectedId = data.project.id; $("project-name").value = data.project.name; $("language").value = data.project.language; $("code").value = data.project.content; renderProjects(); setMessage("Проект создан. Теперь можно писать код.", "success"); } catch (error) { setMessage(error.message, "error"); } finally { setBusy(false); } } async function saveProject() { if (busy) return; if (selectedId === null) { setMessage("Сначала создай проект или выбери существующий.", "error"); return; } const name = $("project-name").value.trim(); if (!name) { setMessage("Название проекта не может быть пустым.", "error"); return; } setBusy(true); try { const data = await api("/api/projects/" + selectedId, { method: "PUT", body: JSON.stringify({ name, language: $("language").value, content: $("code").value }) }); $("project-name").value = data.project.name; await loadProjects(); setMessage("Изменения сохранены на сервере.", "success"); } catch (error) { setMessage(error.message, "error"); } finally { setBusy(false); } } async function removeProject() { if (busy || selectedId === null) { setMessage("Сначала выбери проект для удаления.", "error"); return; } const confirmed = tg?.showConfirm ? await new Promise(resolve => tg.showConfirm( "Удалить выбранный проект без возможности восстановления?", resolve )) : window.confirm("Удалить выбранный проект?"); if (!confirmed) return; setBusy(true); try { await api("/api/projects/" + selectedId, { method: "DELETE" }); clearEditor(); await loadProjects(); setMessage("Проект удалён.", "success"); } catch (error) { setMessage(error.message, "error"); } finally { setBusy(false); } } $("create-btn").addEventListener("click", createNewProject); $("save-btn").addEventListener("click", saveProject); $("reload-btn").addEventListener("click", async () => { if (busy) return; setBusy(true); try { await loadProjects(); setMessage("Список проектов обновлён.", "success"); } catch (error) { setMessage(error.message, "error"); } finally { setBusy(false); } }); $("delete-btn").addEventListener("click", removeProject); async function start() { setBusy(true); try { await loadProjects(); setMessage("Подключение выполнено. Создай первый проект."); } catch (error) { $("connection").textContent = "● Нет подключения"; setMessage(error.message, "error"); } finally { setBusy(false); } } start();
